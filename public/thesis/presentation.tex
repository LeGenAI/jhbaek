\section{Discoveries: Binary Linear Codes}

\begin{frame}{Discovery of Binary Linear Codes (Foundations)}
    \begin{definition}[Binary linear code]
    A \textit{binary linear code} $\mathcal{C}$ of length $n$ and dimension $k$ is a $k$-dimensional subspace of $\mathbb{F}_2^n$. Elements are called \textit{codewords}.
    \end{definition}

    \begin{columns}
        \column{0.5\textwidth}
        \begin{definition}[Generator matrix]
        A $k \times n$ matrix $G$ whose rows form a basis for $\mathcal{C}$. $\mathcal{C} = \{ xG \mid x \in \mathbb{F}_2^k \}$.
        \end{definition}
        
        \begin{definition}[Parity check matrix]
        An $(n-k) \times n$ matrix $H$ such that $c \in \mathcal{C} \iff Hc^T = \mathbf{0}$.
        \end{definition}

        \column{0.5\textwidth}
        \begin{definition}[Standard form]
        $G = [I_k \mid P]$, where $I_k$ is identity and $P$ is $k \times (n-k)$.
        \end{definition}
        
       \begin{definition}[Minimum distance]
        $d(\mathcal{C}) = \min \{ wt(c) \mid c \in \mathcal{C}, c \neq \mathbf{0} \}$.
        \end{definition}
    \end{columns}
    
    \vspace{0.5cm}
    \textbf{Objective}: Discover optimal $[n,k,d]$ codes.
\end{frame}

\section{Discoveries: Self-Orthogonal Embeddings}

\begin{frame}{Self-Orthogonal Embeddings (Foundations)}
    \begin{definition}[Self-orthogonal and Self-dual]
    $\mathcal{C}$ is \textit{self-orthogonal} if $\mathcal{C} \subseteq \mathcal{C}^\perp$. $\mathcal{C}$ is \textit{self-dual} if $\mathcal{C} = \mathcal{C}^\perp$.
    \end{definition}

    \begin{columns}
        \column{0.5\textwidth}
        \begin{definition}[Hull]
        $\Hull(\mathcal{C}) = \mathcal{C} \cap \mathcal{C}^\perp$.
        \end{definition}

        \begin{definition}[Gram matrix condition]
        $\mathcal{C}$ is self-orthogonal $\iff GG^T = \mathbf{0}$.
        \end{definition}

        \column{0.5\textwidth}
         \begin{definition}[SO embedding]
        Minimal self-orthogonal code $\tilde{\mathcal{C}}$ containing $\mathcal{C}$.
        \end{definition}
        
        \begin{definition}[Hermitian inner product]
         $\langle x, y \rangle_H = \sum x_i y_i^q$ over $\mathbb{F}_{q^2}$.
        \end{definition}
    \end{columns}

\end{frame}
\begin{frame}{Self-Orthogonal Embeddings (Algorithm)}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Rank-One Elimination Algorithm}:
        \begin{enumerate}
            \item \textbf{Input}: Generator matrix $G \in \mathbb{F}_q^{k \times n}$.
            \item \textbf{Step 1}: Compute Gram matrix $R = G G^T$.
            \item \textbf{Step 2}: Identify non-zero entry in $R$.
                \begin{itemize} \footnotesize
                    \item If $R_{ii} \ne 0$: Append $v \in \mathbb{F}_q^k$ (from $i$-th row).
                    \item If $R_{ij} \ne 0$: Append $v$ (from pair $i,j$).
                \end{itemize}
            \item \textbf{Step 3}: Update $R \gets R - vv^T$ (cancel entry).
            \item \textbf{Repeat} until $R = \mathbf{0}$.
        \end{enumerate}

        \column{0.5\textwidth}
        \begin{alertblock}{Theorem (Correctness)}
            The algorithm terminates in exactly $r = \rank(G G^T)$ steps. The length of the shortest SO embedding is \textbf{$n + \rank(G G^T)$}.
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}{Self-Orthogonal Embeddings (Results)}
    \textbf{SAT-based Experimentation}:
    \begin{itemize}
        \item While the algebraic algorithm minimizes length, \textbf{SolEvolve} minimizes length \textit{and} maximizes distance ($d$).
    \end{itemize}

\section{Discoveries: Generalized Lucas Cubes}

\begin{frame}{Preliminaries: Interconnection Networks}
    \begin{columns}
        \column{0.6\textwidth}
        \textbf{Interconnection Topologies}:
        modeled by graphs $G=(V,E)$ where $V$ represents processors and $E$ communication links. key metric: $d_G(u,v)$ (shortest path).
        
        \vspace{0.2cm}
        \textbf{Hypercube $Q_n$}:
        \begin{itemize} \footnotesize
            \item Vertices: Binary strings of length $n$.
            \item Edges: Adjacent if strings differ in exactly one coordinate.
            \item Metric: $d_{Q_n}(u,v) = $ Hamming distance.
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Fibonacci \& Lucas Cubes} (Isometric subgraphs of $Q_n$):
        \begin{itemize} \footnotesize
            \item \textbf{Fibonacci Cube $\Gamma_n$}: Induced by strings with no consecutive 1s ($b_i b_{i+1} = 0$).
            \item \textbf{Lucas Cube $\Lambda_n$}: Induced by strings with no consecutive 1s, including wrap-around ($b_1 b_n \neq 1$).

\section{Discoveries: Mutually Orthogonal Latin Squares}

\begin{frame}{Mutually Orthogonal Latin Squares (Foundations)}
    \begin{definition}[Latin Square]
    $n \times n$ array with $n$ symbols, appearing once per row/col.
    \end{definition}
    
    \begin{definition}[MOLS]
    $L_1, L_2$ are orthogonal if every pair $(x,y)$ appears once when superimposed.
    \end{definition}

    \begin{definition}[$k$-net]
    Set of points and lines ensuring unique intersection between parallel classes. Equivalent to MOLS.
    \end{definition}

    \vspace{0.3cm}
    \textbf{Target}: $\text{MOLS}(10,2)$ and $\text{MOLS}(10,3)$ exploration.
\end{frame}
